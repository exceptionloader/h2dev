# MineSMP - LuckPerms Rangsystem
# In der Server-Konsole Zeile fuer Zeile ausfuehren (ohne fuehrenden "/"), Kommentare (#) weglassen.
# Farbschema: Tuerkis (&b/&3), Blau (&9/&1), Weiss (&f), Dunkelgrau (&8)

# --- Gruppen anlegen (Gewicht = Reihenfolge) ---
lp creategroup vip
lp creategroup mvp
lp creategroup legende
lp creategroup builder
lp creategroup supporter
lp creategroup moderator
lp creategroup developer
lp creategroup admin
lp creategroup owner
lp group default setweight 0
lp group vip setweight 10
lp group mvp setweight 20
lp group legende setweight 30
lp group builder setweight 40
lp group supporter setweight 50
lp group moderator setweight 60
lp group developer setweight 70
lp group admin setweight 80
lp group owner setweight 100

# --- Prefixe ---
lp group default meta setprefix 0 "&7"
lp group vip meta setprefix 10 "&b&lVIP &8| &b"
lp group mvp meta setprefix 20 "&3&lMVP &8| &3"
lp group legende meta setprefix 30 "&9&lLEGENDE &8| &9"
lp group builder meta setprefix 40 "&a&lBUILDER &8| &a"
lp group supporter meta setprefix 50 "&e&lSUPPORT &8| &e"
lp group moderator meta setprefix 60 "&6&lMOD &8| &6"
lp group developer meta setprefix 70 "&d&lDEV &8| &d"
lp group admin meta setprefix 80 "&c&lADMIN &8| &c"
lp group owner meta setprefix 100 "&4&lOWNER &8| &4"

# --- Vererbung ---
lp group vip parent add default
lp group mvp parent add vip
lp group legende parent add mvp
lp group builder parent add default
lp group supporter parent add default
lp group moderator parent add supporter
lp group developer parent add moderator
lp group admin parent add moderator
lp group owner parent add admin
